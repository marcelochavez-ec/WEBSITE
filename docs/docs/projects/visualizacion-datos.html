# Lista de meses en orden
meses = ['ENERO', 'FEBRERO', 'MARZO', 'ABRIL', 'MAYO', 'JUNIO',
         'JULIO', 'AGOSTO', 'SEPTIEMBRE', 'OCTUBRE', 'NOVIEMBRE', 'DICIEMBRE']

# 1. Convertir de ancho a largo
df_long = df_ipc.melt(id_vars='AÑOS', value_vars=meses,
                  var_name='MES', value_name='VALOR')

# 2. Crear un orden cronológico para los meses
orden_meses = {mes: i+1 for i, mes in enumerate(meses)}
df_long['MES_NUM'] = df_long['MES'].map(orden_meses)

# 3. Ordenar por año y mes
df_long = df_long.sort_values(by=['AÑOS', 'MES_NUM']).reset_index(drop=True)

# 4. Calcular variación absoluta y porcentual respecto al mes anterior
df_long['VARIACION'] = df_long['VALOR'].diff()
df_long['VARIACION_PORCENTUAL'] = df_long['VARIACION'] / df_long['VALOR'].shift(1) * 100
display(df_long)
